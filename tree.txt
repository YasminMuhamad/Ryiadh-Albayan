users (collection)                  ← كل الطلاب
 └─ userId (document)
     ├─ name
     ├─ name_ar
     ├─ email
     ├─ role          // student
     ├─ subscriptionStatus        // active | cancelled
     ├─ year
     ├─ term
     ├─ profile_pic
     ├─ createdAt
     ├─ coursesCount
     └─ enrollments (subcollection)     ← progress الطالب لكل كورس
         └─ courseId (document)
             ├─ enrolledAt
             ├─ completedLessons: [ lessonId, lessonId ... ]
             ├─ completedLessonsCount
             ├─ quizzesTaken
             ├─ percent
             ├─ status        // in-progress, completed

teachers (collection)
 └─ teacherId (document)
     ├─ name_en
     ├─ name_ar
     ├─ email
     ├─ specialization: "Math" (optional)
     ├─ status: "active" | "deactivated"
     ├─ role: "teacher"
     ├─ profile_pic
     ├─ coursesCount
     └─ createdAt

admins (collection)
 └─ adminId (document)
     ├─ name
     ├─ email
     ├─ role: "admin"
     ├─ profile_pic
     └─ createdAt

courses (collection)
 └─ courseId (document)
     ├─ title_en
     ├─ title_ar
     ├─ description
     ├─ price
     ├─ thumbnail
     ├─ teacherId
     ├─ termId
     ├─ type: "term" | "yearly"
     ├─ category
     ├─ totalLessons
     ├─ totalModules
     ├─ studentsCount
     ├─ revenueTotal
     ├─ avgCompletion
     ├─ avgSatisfaction
     └─ modules (subcollection)
         └─ moduleId (document)
             ├─ title
             └─ lessons (subcollection)
                 └─ lessonId (document)
                     ├─ title
                     ├─ materials: [ { title, file } ]      ← array
                     ├─ videos: [ { title, file } ]         ← array, مسجلة
                     ├─ liveSessions: [                     ← array، لايف
                             { title, dateTime, duration, link, status: scheduled | completed | cancelled } 
                       ]
            ├─ quizzes: [ { id, title, totalMarks } ]
                     ├─ schedule: dateTime (optional)
    └─ reviews: [                         ← array
            { studentId, rating, comment, createdAt } 
    ]

payments (collection)                     ← مستقل، لكل الدفعيات
 └─ paymentId (document)
     ├─ studentId
     ├─ courseId
     ├─ currency
     ├─ amount
     ├─ method
     ├─ type: "one_time" | "subscription"
     ├─ status: "paid" | "pending" | "refunded"
     └─ date

notifications (collection)                ← مستقل، لكل الإشعارات
 └─ notificationId (document)
     ├─ userId
     ├─ title
     ├─ message
     ├─ type: "session" | "general"
     ├─ read: true | false
     └─ createdAt

academic_schedules (collection)          ← مستقل، لكل السنة الدراسية
 └─ scheduleId (document)
     ├─ year
     ├─ term
     ├─ startDate
     └─ endDate

live_sessions (collection)               ← مستقل إذا عايزة تقرير شامل للجلسات
 └─ sessionId (document)
     ├─ courseId
     ├─ teacherId
     ├─ title
     ├─ dateTime
     ├─ duration
     ├─ zoomLink
     └─ status: "upcoming" | "completed" | "cancelled"


analytics (collection)
└─ monthly_revenue (document)
     ├─ yearMonth
     ├─ revenue
     ├─ newStudents
     └─ activeStudents
